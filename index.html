import streamlit as st import sys import io import time # --- APP CONFIG --- st.set_page_config(page_title="Mission: Code Entry", layout="centered") # --- SESSION STATE --- if 'line_idx' not in st.session_state: st.session_state.line_idx = 0 if 'typed_lines' not in st.session_state: st.session_state.typed_lines = [] if 'game_finished' not in st.session_state: st.session_state.game_finished = False # --- ANTI-CHEAT CSS --- # This CSS only disables selection while the game is active. if not st.session_state.game_finished: st.markdown(""" """, unsafe_allow_html=True) else: # Re-enable selection when finished st.markdown("", unsafe_allow_html=True) # --- THE MASTER CODE LIST --- MASTER_CODE = [ "import time", "def slow_print(text):", " for char in text:", " print(char, end='', flush=True)", " time.sleep(0.03)", " print()", "slow_print('--- Welcome to Space Odyssey ---')", "choice1 = input('Investigate or Continue? ').lower()", "if choice1 == 'investigate':", " slow_print('You found an ancient alien beacon.')", "elif choice1 == 'continue':", " slow_print('A pirate corsair jumps from warp!')", "else:", " slow_print('Invalid command. GAME OVER.')" ] # --- UI DISPLAY --- st.title("🚀 Starship Nova: Systems Override") if not st.session_state.game_finished: total = len(MASTER_CODE) current = st.session_state.line_idx target = MASTER_CODE[current] st.progress(current / total) st.info(f"**Step {current + 1} of {total}:** Type the line below exactly.") # We use a code block for the target, but CSS prevents copying it st.code(target, language='python') # Input Box user_input = st.text_input("Enter Code:", key=f"input_{current}", autocomplete="off") if st.button("Check Line"): if user_input == target: st.session_state.typed_lines.append(user_input) if st.session_state.line_idx + 1 < total: st.session_state.line_idx += 1 st.rerun() else: st.session_state.game_finished = True st.rerun() else: st.error("❌ Typo or Indentation Error! (Hint: Check leading spaces)") else: # --- FINAL SUCCESS STAGE --- st.success("✅ Navigation Protocol Restored!") full_code = "\n".join(st.session_state.typed_lines) st.write("### 📋 Instructions for CodeHS") st.write("1. Highlight and copy all the code in the box below.") st.write("2. Paste it into your **CodeHS Python 3 Sandbox**.") # Text area is more reliable for students to copy from than a code block st.text_area("Your Completed Code:", value=full_code, height=300) if st.button("▶️ Test Program (Partial Run)"): output_capture = io.StringIO() sys.stdout = output_capture try: # Note: Interactive 'input' doesn't work in Streamlit, so we simulate a choice exec(full_code, {'time': time, 'input': lambda x: 'continue'}) st.code(output_capture.getvalue()) except Exception as e: st.error(f"Error: {e}") finally: sys.stdout = sys.__stdout__
