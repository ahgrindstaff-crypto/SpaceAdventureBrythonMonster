import streamlit as st import sys import io import time # --- 1. APP CONFIG --- st.set_page_config(page_title="Mission: Code Entry", layout="centered") # --- 2. SESSION STATE (Keep track of progress) --- if 'line_idx' not in st.session_state: st.session_state.line_idx = 0 if 'typed_lines' not in st.session_state: st.session_state.typed_lines = [] if 'game_finished' not in st.session_state: st.session_state.game_finished = False # --- 3. ANTI-CHEAT CSS --- # Only lock the screen if the game isn't finished if not st.session_state.game_finished: st.markdown(""" """, unsafe_allow_html=True) else: st.markdown("", unsafe_allow_html=True) # --- 4. THE MASTER CODE LIST --- MASTER_CODE = [ "import time", "def slow_print(text):", " for char in text:", " print(char, end='', flush=True)", " time.sleep(0.03)", " print()", "slow_print('--- Welcome to Space Odyssey ---')", "choice1 = input('Investigate or Continue? ').lower()", "if choice1 == 'investigate':", " slow_print('You found an ancient alien beacon.')", "elif choice1 == 'continue':", " slow_print('A pirate corsair jumps from warp!')", "else:", " slow_print('Invalid command. GAME OVER.')" ] # --- 5. UI DISPLAY --- st.title("🚀 Starship Nova: Systems Override") if not st.session_state.game_finished: total = len(MASTER_CODE) current = st.session_state.line_idx target = MASTER_CODE[current] st.progress(current / total) st.write(f"**Progress: Line {current + 1} of {total}**") st.info("💡 **Type the following line exactly (watch your spaces!):**") st.code(target, language='python') user_input = st.text_input("Enter Code:", key=f"input_{current}", autocomplete="off") if st.button("Submit Line"): if user_input == target: st.session_state.typed_lines.append(user_input) if st.session_state.line_idx + 1 < total: st.session_state.line_idx += 1 st.rerun() else: st.session_state.game_finished = True st.rerun() else: st.error("❌ Error! Spacing or spelling is off. Try again.") else: st.success("✅ Navigation Protocol Restored!") full_code = "\n".join(st.session_state.typed_lines) st.write("### 📋 Instructions for CodeHS") st.write("Highlight the code below, copy it, and paste it into the CodeHS Sandbox.") # Use a text area for the final copy-paste st.text_area("Your Completed Code:", value=full_code, height=300) if st.button("▶️ Test Program"): output_capture = io.StringIO() sys.stdout = output_capture try: exec(full_code, {'time': time, 'input': lambda x: 'continue'}) st.code(output_capture.getvalue()) except Exception as e: st.error(f"Error: {e}") finally: sys.stdout = sys.__stdout__
